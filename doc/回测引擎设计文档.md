# 回测引擎设计文档

## 设计概述

本项目的核心创新点是采用**Python+C++混合架构**实现高性能回测引擎。参考WonderTrader的生产级架构，采用**C++控制时间循环+Python策略回调**的设计模式，兼顾性能优势和灵活性。

### 核心理念
- **C++引擎**: 控制回测主循环，负责数据管理、订单撮合、持仓管理、绩效计算等高性能计算
- **Python策略**: 用户和AI编写策略代码，通过标准接口与C++引擎交互
- **AI驱动**: 通过LLM自动生成Python策略代码，实现零编码策略设计

### 架构优势
1. **高性能**: C++执行核心回测逻辑，比纯Python快10-100倍
2. **高灵活性**: AI可以生成任意复杂度的策略代码
3. **生产级**: 参考WonderTrader等成熟框架的设计
4. **易于使用**: Python策略接口简单，易于编写和调试

## 策略与回测联动设计

### 核心设计思想

参考WonderTrader的成熟架构，采用**注册-初始化-回放-数据访问-交易执行**五阶段联动模式。

### 阶段1：注册阶段

**目的**: 建立Python策略对象与C++引擎的映射关系

```python
# Python层：创建策略实例
strategy = MAStrategy()

# 调用C++引擎注册
engine = BacktestEngine()
strategy_id = engine.register_strategy(strategy)  # 返回ID: 1
```

```cpp
// C++层：注册逻辑
class BacktestEngine {
public:
    int register_strategy(py::object strategy_obj) {
        int strategy_id = next_strategy_id_++;
        strategies_[strategy_id] = strategy_obj;

        // 注入C++上下文对象
        auto context = std::make_shared<StrategyContext>(this);
        strategy_obj.attr("context") = py::cast(context);

        return strategy_id;
    }

private:
    int next_strategy_id_ = 1;
    std::map<int, py::object> strategies_;
};
```

### 阶段2：初始化阶段

**目的**: C++加载历史数据到内存，策略进行初始化

```python
# Python层：加载数据并初始化
bars = DataServiceAgent().get_daily_data('600000.SH', '20200101', '20241231')
engine.init(bars)  # C++加载数据

# 触发策略初始化回调
engine.call_on_init()
```

```cpp
// C++层：初始化逻辑
void BacktestEngine::init(const std::vector<Bar>& bars) {
    // 加载数据到内存
    bars_ = bars;

    // 初始化持仓
    portfolio_.reset(1000000.0);  // 初始资金100万

    // 初始化时间
    current_bar_index_ = 0;
}

void BacktestEngine::call_on_init() {
    for (auto& [id, strategy] : strategies_) {
        try {
            strategy.attr("on_init")();
        } catch (const py::error_already_set& e) {
            // 处理异常
        }
    }
}
```

### 阶段3：回放阶段

**目的**: C++按时间推进，触发Python策略回调

```cpp
// C++层：回测主循环
void BacktestEngine::run() {
    for (size_t i = 0; i < bars_.size(); i++) {
        const Bar& bar = bars_[i];
        current_bar_ = bar;
        current_bar_index_ = i;

        // 触发所有策略的on_bar回调
        for (auto& [id, strategy] : strategies_) {
            call_python_on_bar(strategy, bar);
        }

        // 处理订单（撮合）
        process_orders(bar);

        // 更新持仓市值
        update_portfolio(bar);
    }

    // 计算绩效指标
    calculate_metrics();
}
```

// C++层：数据访问接口
class StrategyContext {
public:
    // 获取单根K线的某个字段
    double get_bar(const std::string& symbol,
                   const std::string& field,
                   int offset = 0) {
        size_t index = engine_->current_bar_index_ - offset;
        const Bar& bar = engine_->bars_[index];

        if (field == "open") return bar.open;
        if (field == "high") return bar.high;
        if (field == "low") return bar.low;
        if (field == "close") return bar.close;
        if (field == "volume") return bar.volume;

        return 0.0;
    }

    // 获取序列数据
    std::vector<double> get_series(const std::string& symbol,
                                   const std::string& field,
                                   int count) {
        std::vector<double> series;
        size_t start = engine_->current_bar_index_ - count + 1;

        for (size_t i = start; i <= engine_->current_bar_index_; i++) {
            const Bar& bar = engine_->bars_[i];
            double value = get_bar_value(bar, field);
            series.push_back(value);
        }

        return series;
    }

private:
    BacktestEngine* engine_;
};

// C++层：订单处理
class StrategyContext {
public:
    void buy(const std::string& symbol, int quantity, double price) {
        Order order;
        order.order_id = generate_order_id();
        order.symbol = symbol;
        order.direction = "BUY";
        order.quantity = quantity;
        order.price = price;
        order.status = "PENDING";

        // 添加到订单簿
        engine_->add_order(order);
    }

    void sell(const std::string& symbol, int quantity, double price) {
        Order order;
        order.order_id = generate_order_id();
        order.symbol = symbol;
        order.direction = "SELL";
        order.quantity = quantity;
        order.price = price;
        order.status = "PENDING";

        engine_->add_order(order);
    }
};

class BacktestEngine {
public:
    void add_order(const Order& order) {
        pending_orders_.push_back(order);
    }

    void process_orders(const Bar& bar) {
        for (Order& order : pending_orders_) {
            // 模拟撮合（简化版：直接成交）
            if (order.direction == "BUY") {
                double cost = order.price * order.quantity * (1 + commission_rate_);
                if (portfolio_.cash >= cost) {
                    portfolio_.cash -= cost;
                    portfolio_.positions[order.symbol] += order.quantity;
                    order.status = "FILLED";
                } else {
                    order.status = "REJECTED";
                }
            } else {
                // SELL逻辑
                if (portfolio_.positions[order.symbol] >= order.quantity) {
                    portfolio_.cash += order.price * order.quantity * (1 - commission_rate_);
                    portfolio_.positions[order.symbol] -= order.quantity;
                    order.status = "FILLED";
                } else {
                    order.status = "REJECTED";
                }
            }
        }

        // 清理已处理的订单
        auto it = pending_orders_.begin();
        while (it != pending_orders_.end()) {
            if (it->status != "PENDING") {
                it = pending_orders_.erase(it);
            } else {
                ++it;
            }
        }
    }
};

## Python层设计

### 1. 策略基类

**文件**: `src/service_layer/strategy/strategy_base.py`

```python
"""
策略基类 - 所有用户策略必须继承此类
"""

from abc import ABC, abstractmethod
from typing import Dict, Optional, Any

class StrategyBase(ABC):
    """
    策略基类

    用户编写策略时继承此类并实现on_bar方法
    """

    def __init__(self):
        """初始化策略"""
        self.name = self.__class__.__name__
        self.context = None  # C++上下文对象，回测时由引擎注入

    def on_init(self):
        """
        策略初始化回调

        在回测开始前调用一次，可用于：
        - 初始化指标计算器
        - 设置策略参数
        - 预加载必要数据
        """
        pass

    @abstractmethod
    def on_bar(self, context: 'StrategyContext') -> Optional[Dict]:
        """
        每根K线回调（核心方法）

        Args:
            context: C++上下文对象，提供数据访问和交易接口

        Returns:
            订单字典或None，格式为:
            {
                'action': 'buy' | 'sell',
                'symbol': '600000.SH',
                'quantity': 100,
                'price': 10.5
            }
        """
        pass

    def on_order(self, context: 'StrategyContext', order: Dict):
        """
        订单状态回调

        当订单状态发生变化时触发（成交、拒绝、撤销等）

        Args:
            context: C++上下文对象
            order: 订单信息字典
        """
        pass

    def on_trade(self, context: 'StrategyContext', trade: Dict):
        """
        成交回调

        当订单成交时触发

        Args:
            context: C++上下文对象
            trade: 成交信息字典
        """
        pass

    def on_end(self, context: 'StrategyContext'):
        """
        回测结束回调

        在回测结束后调用一次，可用于：
        - 生成策略分析报告
        - 记录策略日志
        - 释放资源
        """
        pass
```

class StrategyAgent:

        # 策略模板

    async def generate_strategy(self,
                               user_request: str,
                               stock_info: Dict[str, Any]) -> Dict[str, Any]:
        """
        生成策略代码

        Args:
            user_request: 用户需求描述
            stock_info: 股票信息（用于策略优化）

        Returns:
            {
                'success': True/False,
                'strategy_code': '...',  # 策略代码
                'strategy_name': 'MAStrategy',
                'description': '策略描述',
                'reasoning': '生成理由'
            }

        # 构建prompt
        prompt = self._build_prompt(user_request, stock_info)

        # 调用LLM生成
        response = await self.llm.ainvoke([HumanMessage(content=prompt)])
        code = response.content

        # 提取代码（如果有markdown标记）
        if '```python' in code:
            code = code.split('```python')[1].split('```')[0].strip()

        return {
            'success': True,
            'strategy_code': code,
            'strategy_name': self._extract_class_name(code),
            'description': f"根据用户需求生成的策略: {user_request}",
            'reasoning': f"LLM根据 '{user_request}' 生成"
        }

        """加载策略模板"""
        return """

1. 继承自StrategyBase类

                'action': 'buy',
                'symbol': '600000.SH',
                'quantity': 100,
                'price': close

        elif ma_short_avg < ma_long_avg:

            if pos > 0:
                return {
                    'action': 'sell',
                    'symbol': '600000.SH',
                    'quantity': -pos,
                    'price': close

        return None
```

- 时间范围: {stock_info.get('start_date', 'N/A')} ~ {stock_info.get('end_date', 'N/A')}

请根据用户需求生成对应的交易策略代码。
只输出Python代码，不要任何其他文字说明。
"""
        return prompt

    def _extract_class_name(self, code: str) -> str:
        """从代码中提取类名"""
        import re
        match = re.search(r'class\s+(\w+)\s*\(', code)
        return match.group(1) if match else "UnknownStrategy"

// src/cpp_engine/include/engine/portfolio.h
#pragma once

#include <map>
#include <vector>
#include "core/position.h"

namespace cpp_engine {

struct Portfolio {
    double cash = 1000000.0;              // 初始资金
    double total_value = 1000000.0;        // 总资产
    double daily_profit_loss = 0.0;        // 当日盈亏
    double total_profit_loss = 0.0;         // 总盈亏
    double total_profit_loss_pct = 0.0;     // 总盈亏比例

    std::map<std::string, Position> positions;  // 持仓表

    void reset(double initial_cash) {
        cash = initial_cash;
        total_value = initial_cash;
        positions.clear();
        total_profit_loss = 0.0;
        total_profit_loss_pct = 0.0;
    }

    void update_market_value(const std::map<std::string, double>& prices) {
        double market_value_sum = 0.0;

        for (auto& [symbol, position] : positions) {
            if (prices.find(symbol) != prices.end()) {
                position.market_value = position.quantity * prices.at(symbol);
                position.profit_loss = (prices.at(symbol) - position.avg_price) * position.quantity;
                if (position.avg_price > 0) {
                    position.profit_loss_pct = position.profit_loss / (position.avg_price * position.quantity);
                }
                market_value_sum += position.market_value;
            }
        }

        total_value = cash + market_value_sum;
        total_profit_loss = total_value - 1000000.0;  // 相对于初始资金
        total_profit_loss_pct = total_profit_loss / 1000000.0;
    }
};

} // namespace cpp_engine
